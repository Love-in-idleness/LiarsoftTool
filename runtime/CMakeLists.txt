cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RScriptRuntime LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    include(CTest)
endif()
set(LIARSOFT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
option(RSCRIPT_BUILD_GAME_MODULES "Build the resource and text modules" ON)
set(RSCRIPT_WINE "" CACHE FILEPATH "Wine executable for Windows cross-built tests")
include_directories(include src "${LIARSOFT_ROOT}/include")

if(RSCRIPT_BUILD_GAME_MODULES AND NOT TARGET liarsofttool_core)
    find_package(Threads REQUIRED)
    find_path(WEBP_INCLUDE_DIR webp/decode.h REQUIRED)
    find_path(VORBIS_INCLUDE_DIR vorbis/vorbisfile.h REQUIRED)
    find_library(WEBP_LIBRARY webp REQUIRED)
    find_library(SHARPYUV_LIBRARY sharpyuv)
    find_library(VORBISFILE_LIBRARY vorbisfile REQUIRED)
    find_library(VORBIS_LIBRARY vorbis REQUIRED)
    find_library(OGG_LIBRARY ogg REQUIRED)
    add_library(liarsofttool_core STATIC
        ${LIARSOFT_ROOT}/src/bigendian.cpp ${LIARSOFT_ROOT}/src/encoding.cpp
        ${LIARSOFT_ROOT}/src/fileio.cpp ${LIARSOFT_ROOT}/src/gscfile.cpp
        ${LIARSOFT_ROOT}/src/transfile.cpp
        ${LIARSOFT_ROOT}/src/gsc_decompiler.cpp ${LIARSOFT_ROOT}/src/archive_index.cpp
        ${LIARSOFT_ROOT}/src/xflarchive.cpp ${LIARSOFT_ROOT}/src/lwg_decoder.cpp
        ${LIARSOFT_ROOT}/src/cg_decompress.cpp ${LIARSOFT_ROOT}/src/wcg_decoder.cpp
        ${LIARSOFT_ROOT}/src/lim_decoder.cpp ${LIARSOFT_ROOT}/src/wav_ogg.cpp
        ${LIARSOFT_ROOT}/src/resource_preview.cpp ${LIARSOFT_ROOT}/src/bmp_normalizer.cpp
        ${LIARSOFT_ROOT}/src/stb_impl.cpp ${LIARSOFT_ROOT}/src/stb_image_impl.cpp)
    target_include_directories(liarsofttool_core PUBLIC "${WEBP_INCLUDE_DIR}" "${VORBIS_INCLUDE_DIR}")
    target_link_libraries(liarsofttool_core PUBLIC "${WEBP_LIBRARY}" "${VORBISFILE_LIBRARY}"
        "${VORBIS_LIBRARY}" "${OGG_LIBRARY}" Threads::Threads)
    if(SHARPYUV_LIBRARY)
        target_link_libraries(liarsofttool_core PUBLIC "${SHARPYUV_LIBRARY}")
    endif()
    if(NOT WIN32)
        find_package(Iconv REQUIRED)
        target_link_libraries(liarsofttool_core PUBLIC Iconv::Iconv)
    endif()
endif()

if(RSCRIPT_BUILD_GAME_MODULES)
    add_library(rscript_resources STATIC src/resource_provider.cpp)
    target_link_libraries(rscript_resources PUBLIC liarsofttool_core)
    add_library(rscript_unicode STATIC src/unicode.cpp)
    target_link_libraries(rscript_unicode PUBLIC liarsofttool_core)
endif()

if(WIN32)
    add_executable(RScriptLauncher WIN32 src/launcher.cpp src/win_util.cpp)
    target_link_libraries(RScriptLauncher PRIVATE comctl32 comdlg32 shell32 ole32 psapi)
    add_library(RScriptHost SHARED src/host.cpp src/win_util.cpp)
    target_link_libraries(RScriptHost PRIVATE psapi shell32)
    add_library(RScriptLogExample SHARED examples/log_plugin.cpp)
    if(MINGW)
        target_link_options(RScriptLauncher PRIVATE -municode)
        foreach(target RScriptLauncher RScriptHost RScriptLogExample)
            target_link_options(${target} PRIVATE -static)
        endforeach()
    endif()
    if(RSCRIPT_BUILD_GAME_MODULES)
        add_library(rscript_imports STATIC src/imports.cpp)
        add_library(RScriptFiles SHARED src/files_plugin.cpp)
        target_link_libraries(RScriptFiles PRIVATE rscript_imports)
        add_library(RScriptResources SHARED src/resources_plugin.cpp)
        target_link_libraries(RScriptResources PRIVATE rscript_resources)
        add_library(RScriptUnicode SHARED src/unicode_plugin.cpp src/engine_text.cpp)
        target_link_libraries(RScriptUnicode PRIVATE rscript_unicode rscript_imports bcrypt shell32 gdi32)
        if(MINGW)
            foreach(target RScriptFiles RScriptResources RScriptUnicode)
                target_link_options(${target} PRIVATE -static)
            endforeach()
        endif()
    endif()
    foreach(target RScriptHost RScriptLogExample RScriptFiles RScriptResources RScriptUnicode)
        if(TARGET ${target})
            set_target_properties(${target} PROPERTIES PREFIX "")
        endif()
    endforeach()
endif()

if(BUILD_TESTING)
    if(RSCRIPT_BUILD_GAME_MODULES)
        find_library(VORBISENC_LIBRARY vorbisenc REQUIRED)
        add_executable(runtime_resource_test tests/resource_test.cpp)
        target_link_libraries(runtime_resource_test PRIVATE rscript_resources "${VORBISENC_LIBRARY}")
        add_executable(runtime_unicode_test tests/unicode_test.cpp)
        target_link_libraries(runtime_unicode_test PRIVATE rscript_unicode)
        foreach(target runtime_resource_test runtime_unicode_test)
            if(MINGW)
                target_link_options(${target} PRIVATE -static)
            endif()
            if(NOT CMAKE_CROSSCOMPILING OR RSCRIPT_WINE)
                add_test(NAME ${target} COMMAND ${RSCRIPT_WINE} $<TARGET_FILE:${target}>)
            endif()
        endforeach()
    endif()
    if(WIN32)
        add_executable(loader_game tests/loader_game.cpp)
        add_library(loader_plugin SHARED tests/loader_plugin.cpp)
        add_library(loader_failure SHARED tests/loader_plugin.cpp)
        target_compile_definitions(loader_failure PRIVATE FIXTURE_FAILURE=1)
        add_library(loader_bad_abi SHARED tests/loader_plugin.cpp)
        target_compile_definitions(loader_bad_abi PRIVATE FIXTURE_ABI=99)
        add_executable(runtime_win_util_test tests/win_util_test.cpp src/win_util.cpp)
        target_link_libraries(runtime_win_util_test PRIVATE psapi)
        if(RSCRIPT_BUILD_GAME_MODULES)
            add_executable(runtime_files_test tests/files_test.cpp)
            add_executable(runtime_imports_test tests/imports_test.cpp)
            target_link_libraries(runtime_imports_test PRIVATE rscript_imports)
            # Manual only: requires a private, independently copied game.
            add_executable(runtime_game_smoke tests/game_smoke.cpp src/win_util.cpp)
            target_link_libraries(runtime_game_smoke PRIVATE psapi gdi32 user32)
            add_executable(runtime_resources_plugin_test tests/resources_plugin_test.cpp)
            target_link_libraries(runtime_resources_plugin_test PRIVATE liarsofttool_core)
            add_executable(runtime_text_test tests/text_test.cpp src/engine_text.cpp)
            target_link_libraries(runtime_text_test PRIVATE rscript_unicode rscript_imports gdi32 user32)
            if(MINGW)
                target_link_options(runtime_files_test PRIVATE -municode -static)
                target_link_options(runtime_imports_test PRIVATE -static)
                target_link_options(runtime_game_smoke PRIVATE -municode -static)
                target_link_options(runtime_resources_plugin_test PRIVATE -static)
                target_link_options(runtime_text_test PRIVATE -static)
            endif()
        endif()
        if(MINGW)
            target_link_options(loader_game PRIVATE -municode)
            foreach(target loader_game loader_plugin loader_failure loader_bad_abi runtime_win_util_test)
                target_link_options(${target} PRIVATE -static)
            endforeach()
        endif()
        foreach(target loader_plugin loader_failure loader_bad_abi)
            set_target_properties(${target} PROPERTIES PREFIX "")
        endforeach()
        if(NOT CMAKE_CROSSCOMPILING OR RSCRIPT_WINE)
            find_package(Python3 COMPONENTS Interpreter REQUIRED)
            set(wine_args)
            if(RSCRIPT_WINE)
                set(wine_args --wine "${RSCRIPT_WINE}")
            endif()
            add_test(NAME runtime_loader COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/loader_test.py" "$<TARGET_FILE_DIR:RScriptLauncher>" ${wine_args})
            add_test(NAME runtime_win_util COMMAND ${RSCRIPT_WINE} $<TARGET_FILE:runtime_win_util_test>)
            if(RSCRIPT_BUILD_GAME_MODULES)
                add_test(NAME runtime_text COMMAND ${RSCRIPT_WINE} $<TARGET_FILE:runtime_text_test>)
                add_test(NAME runtime_imports COMMAND ${RSCRIPT_WINE} $<TARGET_FILE:runtime_imports_test>)
                add_test(NAME runtime_files COMMAND ${RSCRIPT_WINE} $<TARGET_FILE:runtime_files_test>
                    $<TARGET_FILE:RScriptFiles>)
                add_test(NAME runtime_resources_plugin COMMAND ${RSCRIPT_WINE} $<TARGET_FILE:runtime_resources_plugin_test>
                    $<TARGET_FILE:RScriptResources>)
            endif()
        endif()
    endif()
endif()
