cmake_minimum_required(VERSION 3.16)
project(LiarsoftToolGtk4Probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ADWAITA REQUIRED IMPORTED_TARGET libadwaita-1>=1.4)
add_executable(gtk4_probe window.cpp)
target_link_libraries(gtk4_probe PRIVATE PkgConfig::ADWAITA)
enable_testing()
if(WIN32)
    add_test(NAME gtk4_window COMMAND gtk4_probe)
else()
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME gtk4_window COMMAND ${XVFB_RUN} -a $<TARGET_FILE:gtk4_probe>)
    else()
        add_test(NAME gtk4_window COMMAND gtk4_probe)
    endif()
endif()
set_tests_properties(gtk4_window PROPERTIES TIMEOUT 30)
